Template: match against the top

def is_valid(s):
    pairs = {")": "(", "]": "[", "}": "{"}   # closer -> opener, O(1) space
    stack = []
    for c in s:
        if c in pairs:                          # a closer
            if not stack or stack.pop() != pairs[c]:
                return False
        else:
            stack.append(c)
    return not stack                            # nothing left open

Time O(n), space O(n) for the stack ("((((" pushes everything).

Template: monotonic stack ("next warmer day")

def daily_temperatures(temps):
    answer = [0] * len(temps)
    stack = []                                  # indices still waiting
    for i, t in enumerate(temps):
        while stack and temps[stack[-1]] < t:
            j = stack.pop()
            answer[j] = i - j
        stack.append(i)
    return answer

Each index is pushed once and popped at most once, so it's O(n) time despite the inner loop.
